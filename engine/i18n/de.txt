# Words of tTerm for Mac in English: one a line, `key = text` (engine/src/say.rs).
# Every language has the same keys (a check of say.rs sees to it). {name} is a
# value the app puts in; a counting string has a form for each plural of the
# language, "one: … | other: …", and {n} is the count. tTerm, tTerm Web and
# names of terminals and machines are not translated. The page of tTerm Web
# has its own words, in tterm-web: these are the app's alone.

# ------------------------------------------------------------ quit: asked before tTerm quits with terminals open
quit.title = tTerm beenden?
quit.yes = Beenden
quit.cancel = Abbrechen
quit.shells_end = one: Die Shell von {n} Terminal wird beendet. | other: Die Shells von {n} Terminals werden beendet.
# {running} lists what runs: "npm run dev in P/web-01, vim in P/db".
quit.running = one: Läuft gerade: {running}. Das wird gestoppt, und die Shell von {n} Terminal wird beendet. | other: Läuft gerade: {running}. Das wird gestoppt, und die Shells von {n} Terminals werden beendet.
quit.running_item = {name} in {label}
# How the language lists them, two at a time: "a, b" ("a、b" in Japanese).
quit.list = {a}, {b}
# Two sentences one after the other: "a b" ("ab" in Chinese and Japanese).
quit.both = {first} {second}
quit.web_end = one: {n} Terminal, das von tTerm Web aus auf diesem Computer geöffnet wurde, wird beendet. | other: {n} Terminals, die von tTerm Web aus auf diesem Computer geöffnet wurden, werden beendet.
